local factory=assert(loadfile(arg[1]))()
local function close(actual,expected,tolerance)
    assert(math.abs(actual-expected)<(tolerance or 0.000001),tostring(actual).." ~= "..tostring(expected))
end
local function position(height,horizontal) return {x=horizontal or 0,y=5,z=height} end
local focus=factory()
close(focus.follow("player",position(10),0).z,10)
for _,now in ipairs({0.1,0.3,0.59}) do
    local center=focus.follow("player",position(11,now),now)
    close(center.z,10);close(center.x,now);close(center.y,5)
end
close(focus.follow("player",position(10),0.6).z,10)
close(focus.follow("player",position(10),1.2).z,10)
close(focus.follow("player",position(9),1.3).z,10)
close(focus.follow("player",position(10),1.7).z,10)
close(focus.follow("player",position(10),2).z,10)

local function sample(rate)
    local current=factory()
    current.follow("player",position(0),0)
    current.follow("player",position(2),1)
    current.follow("player",position(2),1.5)
    local result
    for index=1,rate/2 do result=current.follow("player",position(2),1.5+index/rate) end
    return result.z
end
close(sample(30),sample(60));close(sample(60),sample(144))
focus=factory()
focus.follow("player",position(0),0)
local slope
for index=1,180 do slope=focus.follow("player",position(index/60),index/60) end
assert(slope.z>2.8 and slope.z<3)
